% -- preamble for instructions
% --
% -- sets up document and defines macros to generate images
% -- for instruction explanation blocks

\documentclass
[
    preview,
    border=0pt
    % varwidth
]
{standalone}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{xcolor}
\usepackage{graphicx}
\usepackage{tcolorbox}
\usepackage{url}
\usepackage[margin=1.2cm]{geometry}

% font selection
\usepackage{courierten}
\renewcommand*\familydefault{\ttdefault}
\frenchspacing

% fbox settings
\renewcommand{\fboxrule}{\dimexpr 0.3pt}

% color definitions
\definecolor{asm-blue}  {HTML}{9CD0FF}
\definecolor{asm-green} {HTML}{CCFF99}
\definecolor{asm-grey}  {HTML}{E8E8F2}
\definecolor{asm-orange}{HTML}{FFCC99}
\definecolor{asm-pink}  {HTML}{FFCCFF}
\definecolor{asm-purple}{HTML}{CCCCFF}
\definecolor{asm-red}   {HTML}{FF9999}
\definecolor{asm-yellow}{HTML}{FFFFCC}

% instruction block
\newenvironment{instructionblock}[1]
{%
    \begin{minipage}[t]{#1}
    \vspace{0pt}
    \begin{tcolorbox}
    [
        sharp corners,
        colback=asm-yellow,
        colframe=black,
        boxrule=0.5pt,
        beforeafter skip=0cm,
        leftright skip=0cm,
    ]
}
{
\end{tcolorbox}
\end{minipage}
}

% bit representation of an instruction
\newcommand{\binaryoperation}[2]
{
    \\
    \mbox{#1}
    \hfill\textbf{#2}
    \\
}

% colorboxes for bit representation of instructions
% #1: background color
% #2: bits as string
\newcommand{\bitgroup}[2]
{%
\fcolorbox{black}{#1}{\strut #2}%
}

% constant definitions
\newcommand{\imagewidth}{\dimexpr 9cm}
