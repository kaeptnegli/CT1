\begin{instructionblock}{1.4\imagewidth}%
\textbf{msr apsr, rn}\hfill\textit{T1}\\
    rn $\in$ \extendedregisters\\
    \binaryoperation{%
        \bitgroup{asm-blue}     {11110}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {1110}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-pink}     {nnnn}%
        \bitgroup{asm-grey}     {10}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {10}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-red}      {00000000}%
    }%
    {}\\
    apsr = rn;\vspace{0.5em}

    \itshape
    `\textbf{msr}` can move other special registers, but we use only
    \textbf{apsr}.
    This would change the red coloured bits.
\end{instructionblock}
