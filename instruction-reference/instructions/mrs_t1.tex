\begin{instructionblock}{1.4\imagewidth}%
\textbf{mrs rd, apsr}\hfill\textit{T1}\\
    rd $\in$ \extendedregisters\\
    \binaryoperation{%
        \bitgroup{asm-blue}     {11110}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {1111}%
        \bitgroup{asm-grey}     {1}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {1111}%
        \bitgroup{asm-grey}     {10}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-grey}     {0}%
        \bitgroup{asm-green}    {dddd}%
        \bitgroup{asm-red}      {00000000}%
    }%
    {}\\
    rd = apsr;\vspace{0.5em}

    \itshape
    `\textbf{mrs}` can move other special registers, but we use only
    \textbf{apsr}.
    This would change the red coloured bits.
\end{instructionblock}
